262 هـ -٣ حَدَّثَنَا مَحْمُودُ بْنُ عُيَيْنَانِ ، أَخْبَرَنَا حَفْصُ بْنُ عُمَرَ بْنِ عُبَيْدٍ ) بالتَّصْغِيرِ ( هُوَ الطَّنَافِسِيُّ ) - بفتح الطاء ، و بكسر الفاء ، و فتحها ( ١ ، البساط الذي له حمل(٢) و حصير من سعف قدره ذراع (٣) ، و كَأْنَ (٤) النسبة للعمل ، أو البيع إِشْعَارًا بأنَّه صار عَلِمًا له بالغلبة ، و اشتهر به ، و هو ثقفة ، كذا ذكره الشراح (٤)(٥) ، و في نسخة ضعيفة هو الطُّفَالِيّ - بضم الطاء ، و بالفاء ، آخره لام ، بعده تحْتِية [ مشدَّة -٦] (٧) ( أَخْبَرَنَا ) و في بعض النسخ أَئِبَّنَا ( زُهَيْرٌ ) بضم زاي ، و فتح هاء ( أَبُو خَيْثَمَة ) بتحْتِية ساكنة بين فتح معجمة ، و مثَلَّة (٨) و احْتَرَزَ به عن زهير أبي المنذر (٩) ؛ لأنَّه غير موثوق به (١٠) ( عَنْ حُمَيْدٍ )- بالتَّصْغِيرِ - أي الطويل (١) بكسر الطاء و الفاء ، و بضمهما ، و بكسر الطاء و فتح الفاء ، و بالعكس . انظر: النهاية ٣/١١٥ ، القاموس المحيط ص ٥١٣ ، تاج العروس ١٦/٢١ : مادة طنفس . (٢) لسان العرب : مادة طنفس ٦/٢٧ (٣) الذراع من الإنسان : من المرْفَق إلى أطراف الأصابع ، و الذراع الشرعية تبلغ ٨٧٥ سم. انظر: المصابح المنير : مادة ذرع ص ٧٩ ، المكابيل و الموازين الإسلامية لفاترهنش ص ٨٣ (٤) قاله العِصَام في شرحه لشمائل النبي صلى الله عليه وسلم . (٥) ( ت ) الكُوفِيّ ، من العاشرة . (٦) تقريب التهذيب ١/١٨٦-(ت ١٥٤٩) ( ت ) ح د س ق ) زُهَيْر بن حرب بن شَدَّادَ النَّسَائِيُّ ، نزيل بغداد ، ثقة ، ثبت ، روى عنه مُسْلِم أكثر من ألف حديث ، من العاشرة ، مات سنة ٣٤٣هـ ، و هو ابن ٧٤ سنة . (٧) تقريب التهذيب ١/٢٥٨-(ت ٢٢٣١) ( ع ) زهير بن مُحَمَّد التنميمي ، أبو المنذر الخراساني ، سكن الشام ، ثم الحجاز ، ثقة ، إلا أن رواية أهل الشام عنه غير مستقيمة ، فضُعِّفَ بسببها ، قالَ البُخَارِي : عَن أَحْمَد ، كأن زهيرا الذي يروي عنه الشاميون آخر ، و قالَ أَبُو حاتم : حدث بالشام من حفظه ، فكتَّر غلطه ، من السابعة ، مات سنة ٦٢ هـ . تقريب التهذيب ٢/٣٤٢-(ت ٢٠٦٠) بتحقيق أبي الأشبال .
