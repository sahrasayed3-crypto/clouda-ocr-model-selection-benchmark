٢٧٦ يُخَطِر المجمع بين محارم، والخنثى ٢٧٧ شروط الولي ٢٧٨ الولي المخير ٢٧٩ كفاءة الزوج وأن لا يكون عدواً ٢٨٠ تعريف البكر واجبارها ٢٨٢ ابتداء انتداب البكر ولو من مجب ٢٨٢ استنان الولي غير المجبب لمكير ٢٨٢ إذن البيت يجب التصريح به ٢٨٣ البينة البكر كالثيب في الأذن ٢٨٣ مذهب النعمان أربع في شأنها ٢٨٤ الولي الأولياء الأب ٢٨٤ تزييب الأولياء ٢٨٥ لا حق لابن أن يكون إباً إذا لم يكن له صفة ترضحه ٢٨٦ تقديم ولي الكل الولي المجبب ٢٨٦ لو كبر غير المجبب لا يُد له تقدم إذنها ٢٨٧ انتقال الولي إلى الابعث عند وجود أعذار ٢٨٨.٢٨٧ سواب الولاة ٢٨٨ مسائل يجزع فيها الحكام ٢٨٨ بعقد الولي أو عليه يزج القاضي ونائبه ٢٩٠ وكذا إذا حبس الولي أو أحرم أو تعز أو احتفى ٢٩١ يزج القاضي الولي إذا أراد نكاحها وكذا المحجور على إيلها والموقوفة ٢٩٢ يزج القاضي المجزعة ٢٩٢ يزج بات القاضي أو السلطان القاضي ٢٩٣ يستني القاضي بإذن السلطان ٢٩٣ بطلت اجتماع الأولياء إذا قالت زوجني ٢٩٤ صحة توكيل الأولياء واحداً ٢٩٦ لا يشترط إثبات قولها مطلقة عند حاكم ٢٩٦ تجري الولياء عند وجود ريبة ٢٩٧ أقسام الطلاق خمسة ٢٩٨ سؤال العاقد عن الخلع والطلاق والتعاليق ٢٩٩ الخلع وأركانه وصيغه
